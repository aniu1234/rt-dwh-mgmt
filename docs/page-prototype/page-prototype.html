-- Source: MySQL CDC
CREATE TABLE mysql_orders_source (
  order_id BIGINT,
  user_id BIGINT,
  amount DECIMAL(10,2),
  order_time TIMESTAMP(3),
  PRIMARY KEY (order_id) NOT ENFORCED
) WITH (
  'connector' = 'mysql-cdc',
  'hostname' = '192.168.1.10',
  'port' = '3306',
  'database-name' = 'shop',
  'table-name' = 'orders'
);

-- Target: Paimon ODS
CREATE TABLE paimon_ods_orders (
  order_id BIGINT,
  user_id BIGINT,
  amount DECIMAL(10,2),
  order_time TIMESTAMP(3),
  order_date STRING,
  PRIMARY KEY (order_id, order_date) NOT ENFORCED
) WITH (
  'connector' = 'paimon',
  'path' = 'oss://data/paimon/ods.db/ods_orders',
  'metastore' = 'hive'
);

INSERT INTO paimon_ods_orders
SELECT order_id, user_id, amount, order_time,
  DATE_FORMAT(order_time, 'yyyy-MM-dd') AS order_date
FROM mysql_orders_source;
